def login()
    Sistem autentikasi sebelum akses menu
    clear_screen()
    print(fn{B}=== NEMESIS-X SYSTEM ACCESS ==={W})
    input_code = input(f{Y}Masukkan Access Code {W})
    
    if input_code == ACCESS_CODE
        print(fn{G}[+] Akses Diterima. Mengautentikasi...{W})
        time.sleep(1.5)
        clear_screen() # Membersihkan layar segera setelah sukses login
        return True
    else
        print(fn{R}[!] KODE SALAH. Akses Ditolak.{W})
        time.sleep(2)
        return False

def scan_port(target, port)
    Logika pengecekan port menggunakan socket
    try
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.settimeout(0.6) # Timeout seimbang antara akurasi dan kecepatan
        result = s.connect_ex((target, port))
        if result == 0
            print(f  {G}[OPEN]{W} Port {B}{port}{W} terdeteksi terbuka)
        s.close()
    except Exception
        pass
